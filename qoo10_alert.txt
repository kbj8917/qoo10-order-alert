#!/usr/bin/env python3
"""Qoo10 Japan 신규 주문 알림 (텔레그램 / 디스코드)

외부 라이브러리 없이 파이썬 3.8+ 표준 라이브러리만 사용합니다.

사용법:
  python qoo10_alert.txt            # 한 번 확인 (GitHub Actions용)
  python qoo10_alert.txt --loop     # 계속 실행 (PC용, 기본 3분 간격)
  python qoo10_alert.txt --test     # API 연결 테스트 (주문 원본 데이터 출력)
  python qoo10_alert.txt --chat-id  # 텔레그램 chat_id 확인
"""
import argparse
import json
import os
import sys
import time
import urllib.parse
import urllib.request
from datetime import datetime, timedelta, timezone

API_BASE = "https://api.qoo10.jp/GMKT.INC.Front.QAPIService/ebayjapan.qapi"
JST = timezone(timedelta(hours=9))

# ShippingStat: 1=배송대기(入金待ち/配送待ち) 2=배송요청(신규 결제완료) 3=발송확인 4=배송중 5=배송완료
STATUS_NAMES = {"1": "배송대기", "2": "배송요청(신규)", "3": "발송확인", "4": "배송중", "5": "배송완료"}


def env(name, default=""):
    return os.environ.get(name, default).strip()


def load_dotenv(path=".env"):
    """같은 폴더의 .env 파일을 읽어 환경변수로 설정 (PC 실행용)."""
    if not os.path.exists(path):
        return
    with open(path, encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if not line or line.startswith("#") or "=" not in line:
                continue
            k, v = line.split("=", 1)
            os.environ.setdefault(k.strip(), v.strip().strip('"').strip("'"))


# ---------------------------------------------------------------- Qoo10 API
def qapi(method, key, **params):
    query = {"v": "1.0", "returnType": "json", "method": method, "key": key}
    query.update({k: v for k, v in params.items() if v is not None})
    url = f"{API_BASE}?{urllib.parse.urlencode(query)}"
    req = urllib.request.Request(url, headers={
        "GiosisCertificationKey": key,
        "QAPIVersion": "1.0",
        "User-Agent": "qoo10-order-alert/1.0",
    })
    with urllib.request.urlopen(req, timeout=30) as r:
        body = r.read().decode("utf-8-sig")
    try:
        data = json.loads(body)
    except json.JSONDecodeError:
        raise RuntimeError(f"Qoo10 응답을 해석할 수 없습니다: {body[:300]}")
    code = str(data.get("ResultCode", data.get("resultCode", "0")))
    if code not in ("0", "-10001"):  # -10001: 조회 결과 없음
        raise RuntimeError(f"Qoo10 API 오류 [{method}] code={code} msg={data.get('ResultMsg')}")
    return data


def get_seller_key():
    sak = env("QOO10_SAK")
    if sak:
        return sak
    api_key, uid, pwd = env("QOO10_API_KEY"), env("QOO10_USER_ID"), env("QOO10_PASSWORD")
    if not (api_key and uid and pwd):
        sys.exit("설정 오류: QOO10_SAK 또는 (QOO10_API_KEY + QOO10_USER_ID + QOO10_PASSWORD)가 필요합니다.")
    data = qapi("CertificationAPI.CreateCertificationKey", api_key, user_id=uid, pwd=pwd)
    sak = data.get("ResultObject")
    if not sak or not isinstance(sak, str):
        raise RuntimeError(f"판매자 인증키 발급 실패: {data}")
    return sak


def field(o, *names, default=""):
    """응답 필드명 대소문자 차이를 흡수."""
    low = {k.lower(): v for k, v in o.items()}
    for n in names:
        v = low.get(n.lower())
        if v not in (None, ""):
            return v
    return default


def fetch_orders(sak):
    days = int(env("LOOKBACK_DAYS", "3") or 3)
    statuses = [s.strip() for s in env("QOO10_STATUSES", "1,2").split(",") if s.strip()]
    now = datetime.now(JST)
    sdate, edate = (now - timedelta(days=days)).strftime("%Y%m%d"), now.strftime("%Y%m%d")
    orders = {}
    for st in statuses:
        data = qapi("ShippingBasic.GetShippingInfo_v2", sak, ShippingStat=st,
                    search_Sdate=sdate, search_Edate=edate, search_condition="1")
        for o in data.get("ResultObject") or []:
            no = str(field(o, "orderNo", "OrderNo"))
            if no:
                o["_status"] = st
                orders[no] = o
    return orders


# ---------------------------------------------------------------- 알림
def translate_ja_to_ko(text):
    """일본어 상품명/옵션을 한국어로 번역 (실패 시 빈 문자열 반환, 알림 누락 방지)."""
    s = (text or "").strip()
    if not s:
        return ""
    # 1) Google Translate gtx
    try:
        q = urllib.parse.urlencode({"client": "gtx", "sl": "ja", "tl": "ko", "dt": "t", "q": s})
        req = urllib.request.Request(
            f"https://translate.googleapis.com/translate_a/single?{q}",
            headers={"User-Agent": "Mozilla/5.0"},
        )
        with urllib.request.urlopen(req, timeout=8) as r:
            data = json.loads(r.read().decode("utf-8"))
        out = "".join(seg[0] for seg in (data[0] or []) if seg and seg[0]).strip()
        if out and out != s:
            return out
    except Exception:
        pass
    # 2) MyMemory 폴백
    try:
        q = urllib.parse.urlencode({"q": s[:450], "langpair": "ja|ko"})
        req = urllib.request.Request(
            f"https://api.mymemory.translated.net/get?{q}",
            headers={"User-Agent": "Mozilla/5.0"},
        )
        with urllib.request.urlopen(req, timeout=8) as r:
            data = json.loads(r.read().decode("utf-8"))
        out = ((data.get("responseData") or {}).get("translatedText") or "").strip()
        if out and out != s:
            return out
    except Exception:
        pass
    return ""


def _bundle_key(o):
    pack = str(field(o, "packNo", "PackNo") or "").strip()
    if pack and pack != "0":
        return f"pack:{pack}", f"장바구니 {pack}"
    buyer = str(field(o, "buyer", "Buyer") or "").strip()
    recv = str(field(o, "receiver", "Receiver", "receiverName", "ReceiverName") or "").strip()
    if buyer and recv:
        return f"person:{buyer}|{recv}", f"{buyer} / {recv}"
    if buyer:
        return f"buyer:{buyer}", buyer
    return "", ""


def find_combined_map(new_orders, all_orders):
    """전체 미처리 주문 중 같은 장바구니(packNo) 또는 동일 구매자/수령인 2건 이상 합배송 탐지."""
    buckets = {}
    labels = {}
    for o in all_orders.values():
        k, lbl = _bundle_key(o)
        if not k:
            continue
        buckets.setdefault(k, []).append(str(field(o, "orderNo", "OrderNo")))
        labels[k] = lbl
    new_nos = {str(field(o, "orderNo", "OrderNo")) for o in new_orders}
    order_bundle = {}
    summary_lines = []
    for k, nos in buckets.items():
        if len(nos) >= 2 and any(n in new_nos for n in nos):
            for n in nos:
                order_bundle[n] = len(nos)
            summary_lines.append(f"  • {labels[k]} — {len(nos)}건 ({', '.join(nos)})")
    return order_bundle, summary_lines


def format_order(o, bundle_count=0):
    title = field(o, "itemTitle", "ItemTitle")
    opt = field(o, "option", "Option", "optionCode")
    qty = field(o, "orderQty", "OrderQty")
    total = field(o, "total", "Total", "orderPrice", "OrderPrice")
    title_ko = translate_ja_to_ko(title)
    opt_ko = translate_ja_to_ko(opt) if opt else ""
    header = f"🛒 Qoo10 신규 주문! 📮[합배송 {bundle_count}건]" if bundle_count >= 2 else "🛒 Qoo10 신규 주문!"
    lines = [
        header,
        f"주문번호: {field(o, 'orderNo', 'OrderNo')}",
        f"주문일시: {field(o, 'orderDate', 'OrderDate')}",
        f"상품(한글): {title_ko}" if title_ko else f"상품: {title}",
    ]
    if title_ko:
        lines.append(f"상품(원문): {title}")
    if opt:
        if opt_ko:
            lines.append(f"옵션(한글): {opt_ko}")
            lines.append(f"옵션(원문): {opt}")
        else:
            lines.append(f"옵션: {opt}")
    lines.append(f"수량: {qty}")
    if total:
        lines.append(f"금액: ¥{total}")
    buyer = field(o, "buyer", "Buyer")
    if buyer:
        lines.append(f"구매자: {buyer}")
    lines.append(f"상태: {STATUS_NAMES.get(o.get('_status'), o.get('_status'))}")
    return "\n".join(lines)


def post(url, payload, as_json=False):
    if as_json:
        data, ctype = json.dumps(payload).encode(), "application/json"
    else:
        data, ctype = urllib.parse.urlencode(payload).encode(), "application/x-www-form-urlencoded"
    req = urllib.request.Request(url, data=data, headers={"Content-Type": ctype, "User-Agent": "qoo10-order-alert/1.0"})
    with urllib.request.urlopen(req, timeout=30) as r:
        return r.read()


def notify(text):
    sent = False
    tok, chat = env("TELEGRAM_BOT_TOKEN"), env("TELEGRAM_CHAT_ID")
    if tok and chat:
        post(f"https://api.telegram.org/bot{tok}/sendMessage", {"chat_id": chat, "text": text})
        sent = True
    hook = env("DISCORD_WEBHOOK_URL")
    if hook:
        post(hook, {"content": text}, as_json=True)
        sent = True
    if not sent:
        print("[알림 채널 미설정] " + text.replace("\n", " | "))


# ---------------------------------------------------------------- 상태 저장
def load_state(path):
    if not os.path.exists(path):
        return None
    with open(path, encoding="utf-8") as f:
        return json.load(f)


def save_state(path, state):
    # 최근 2000건만 보관
    state["seen"] = state["seen"][-2000:]
    with open(path, "w", encoding="utf-8") as f:
        json.dump(state, f, ensure_ascii=False)


def log(msg):
    if not os.environ.get("GITHUB_ACTIONS"):
        print(msg)


def _parse_dt(val):
    s = str(val or "").strip()
    if not s:
        return None
    for fmt in ("%Y-%m-%d %H:%M:%S", "%Y-%m-%d %H:%M", "%Y-%m-%d", "%Y/%m/%d %H:%M:%S", "%Y/%m/%d", "%Y%m%d"):
        try:
            return datetime.strptime(s[:19], fmt).replace(tzinfo=JST)
        except ValueError:
            continue
    return None


def find_delay_warnings(all_orders, now_jst=None):
    """배송요청(상태 2) 중 발송예정일 임박/경과 또는 주문 후 48시간 이상 경과한 주의 건 탐지."""
    now = now_jst or datetime.now(JST)
    today_date = now.date()
    warn_nos = []
    for o in all_orders.values():
        if str(o.get("_status") or "") != "2":
            continue
        no = str(field(o, "orderNo", "OrderNo") or "").strip()
        if not no:
            continue
        est = _parse_dt(field(o, "estShippingDate", "EstShippingDate", "shippingDate", "ShippingDate"))
        if est is not None:
            if est.date() <= today_date:
                warn_nos.append(no)
            continue
        odt = _parse_dt(field(o, "orderDate", "OrderDate", "paymentDate", "PaymentDate"))
        if odt is not None and (now - odt) >= timedelta(hours=48):
            warn_nos.append(no)
    return warn_nos


def format_summary(new_orders, total_orders_count, bundle_lines=None, warn_nos=None):
    """신규 주문들을 상품·옵션별로 묶어 발주 수량을 한눈에 보기 쉽게 정리."""
    groups = {}
    order_seq = []
    for o in new_orders:
        code = str(field(o, "itemCode", "ItemCode") or "").strip()
        title = str(field(o, "itemTitle", "ItemTitle") or "").strip()
        opt = str(field(o, "option", "Option", "optionCode") or "").strip()
        key = code or title or "상품"
        if key not in groups:
            groups[key] = {"title": title, "count": 0, "qty": 0, "opts": {}, "opt_seq": []}
            order_seq.append(key)
        g = groups[key]
        g["count"] += 1
        try:
            q = int(field(o, "orderQty", "OrderQty", default=1) or 1)
        except (TypeError, ValueError):
            q = 1
        g["qty"] += q
        if opt:
            if opt not in g["opts"]:
                g["opts"][opt] = 0
                g["opt_seq"].append(opt)
            g["opts"][opt] += q

    total_qty = sum(g["qty"] for g in groups.values())
    warn_list = warn_nos or []
    lines = [
        f"📦 Qoo10 신규 {len(new_orders)}건 / 발주 총 {total_qty}개 (미처리 {total_orders_count}건 · 주의 {len(warn_list)}건)",
        "",
    ]
    for key in order_seq:
        g = groups[key]
        title_ko = translate_ja_to_ko(g["title"])
        name = title_ko or g["title"]
        lines.append(f"• {name} — {g['qty']}개 발주")
        for opt in g["opt_seq"]:
            opt_ko = translate_ja_to_ko(opt)
            opt_label = opt_ko or opt
            lines.append(f"  └ {opt_label}: {g['opts'][opt]}개")
    if bundle_lines:
        lines.append("")
        lines.append("📮 합배송(묶음) 있음:")
        lines.extend(bundle_lines)
    if warn_list:
        lines.append("")
        shown = ", ".join(warn_list[:10]) + (" 외" if len(warn_list) > 10 else "")
        lines.append(f"⚠️ 발송지연 주의 {len(warn_list)}건: {shown}")
    return "\n".join(lines)


def check_once(sak):
    path = env("STATE_FILE", "seen_orders.json")
    state = load_state(path)
    orders = fetch_orders(sak)
    now = datetime.now(JST).strftime("%Y-%m-%d %H:%M")
    if state is None:
        # 첫 실행: 기존 주문은 알림 없이 기록만
        save_state(path, {"seen": list(orders.keys())})
        notify(f"✅ Qoo10 주문 알림이 시작되었습니다.\n현재 처리 대기 주문: {len(orders)}건\n이제부터 새 주문이 들어오면 알려드릴게요.")
        log(f"[{now}] 첫 실행 - 기존 주문 {len(orders)}건 기록")
        return
    seen = set(state["seen"])
    new = [o for no, o in orders.items() if no not in seen]
    if new:
        sorted_new = sorted(new, key=lambda x: str(field(x, "orderDate", "OrderDate")))
        order_bundle, bundle_lines = find_combined_map(sorted_new, orders)
        warn_nos = find_delay_warnings(orders)
        notify(format_summary(sorted_new, len(orders), bundle_lines=bundle_lines, warn_nos=warn_nos))
        for o in sorted_new:
            no = str(field(o, "orderNo", "OrderNo"))
            notify(format_order(o, bundle_count=order_bundle.get(no, 0)))
            state["seen"].append(no)
    save_state(path, state)
    log(f"[{now}] 조회 {len(orders)}건 / 신규 {len(new)}건")


def load_actions_secrets():
    """GitHub Actions: S 환경변수(JSON)에 담긴 Secrets를 환경변수로 풀기."""
    raw = os.environ.get("S")
    if not raw:
        return
    for k, v in json.loads(raw).items():
        if isinstance(v, str):
            os.environ.setdefault(k, v)


def main():
    load_dotenv()
    load_actions_secrets()
    ap = argparse.ArgumentParser(description="Qoo10 Japan 신규 주문 알림")
    ap.add_argument("--loop", action="store_true", help="계속 실행")
    ap.add_argument("--interval", type=int, default=int(env("INTERVAL_MINUTES", "3") or 3), help="확인 간격(분)")
    ap.add_argument("--test", action="store_true", help="API 연결 테스트")
    ap.add_argument("--chat-id", action="store_true", help="텔레그램 chat_id 확인")
    a = ap.parse_args()

    if a.chat_id:
        tok = env("TELEGRAM_BOT_TOKEN") or sys.exit("TELEGRAM_BOT_TOKEN을 먼저 설정하세요.")
        with urllib.request.urlopen(f"https://api.telegram.org/bot{tok}/getUpdates", timeout=30) as r:
            res = json.load(r).get("result", [])
        ids = {u["message"]["chat"]["id"]: u["message"]["chat"].get("first_name", "") for u in res if "message" in u}
        if not ids:
            print("메시지가 없습니다. 텔레그램에서 내 봇에게 아무 메시지나 보낸 후 다시 실행하세요.")
        for i, n in ids.items():
            print(f"chat_id = {i}  ({n})")
        return

    sak = get_seller_key()
    if a.test:
        orders = fetch_orders(sak)
        print(f"✅ API 연결 성공 - 조회된 주문 {len(orders)}건")
        for o in list(orders.values())[:3]:
            print(json.dumps(o, ensure_ascii=False, indent=2))
        notify("🔔 Qoo10 주문 알림 테스트 메시지입니다.")
        return

    if not a.loop:
        check_once(sak)
        return
    print(f"모니터링 시작 ({a.interval}분 간격, 종료: Ctrl+C)")
    while True:
        try:
            check_once(sak)
        except Exception as e:  # 네트워크 오류 등은 다음 주기에 재시도
            print(f"오류: {e}")
            if "인증" in str(e) or "-1" in str(e):
                try:
                    sak = get_seller_key()
                except Exception:
                    pass
        time.sleep(a.interval * 60)


if __name__ == "__main__":
    main()
